SELECT 
    o.shipping_city,
    SUM(p.amount) AS Total_Sales
FROM dbo.orders_details o
INNER JOIN dbo.payment_details p
    ON o.order_id = p.order_id
GROUP BY o.shipping_city
ORDER BY Total_Sales DESC;

SELECT
    p.category,
    SUM(oi.quantity * oi.unit_price * (1 - oi.discount_percent / 100.0)) AS Total_Sales
FROM dbo.order_item_details oi
INNER JOIN dbo.product_details p
    ON oi.product_id = p.product_id
GROUP BY p.category
ORDER BY Total_Sales DESC;


SELECT
    status,
    COUNT(DISTINCT order_id) AS Total_Orders
FROM dbo.orders_details
GROUP BY status
ORDER BY Total_Orders DESC;


SELECT
    payment_method,
    SUM(amount) AS Total_Sales
FROM dbo.payment_details
GROUP BY payment_method
ORDER BY Total_Sales DESC;

SELECT
    p.category,
    SUM(oi.quantity) AS Total_Quantity
FROM dbo.order_item_details oi
INNER JOIN dbo.product_details p
    ON oi.product_id = p.product_id
GROUP BY p.category
ORDER BY Total_Quantity DESC;

    reason,
    COUNT(*) AS Total_Returns
FROM dbo.return_details
GROUP BY reason
ORDER BY Total_Returns DESC;
